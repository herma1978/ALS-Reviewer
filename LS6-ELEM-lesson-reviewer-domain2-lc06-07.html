      One-time setup: Results & Dashboard sheet

      1. Create a new Google Sheet for results (separate from your private lesson-planning sheet).
      2. Open Extensions → Apps Script, delete any starter code, and paste the script below.
      3. Leave SYNC_KEY exactly as it is in the script below — it already matches this Reviewer file, which is how every student's copy connects automatically without them ever typing a key. Changing it will quietly break syncing for every student device that uses the simple "paste the link" box, since they have no way to enter a different key themselves. (If you deliberately want a different key, you must also open the hidden teacher Settings on every student device and set the matching Sync Key there — not practical for a class, so this is not recommended.)
      4. Click Deploy → New deployment → Web app. "Execute as" Me, "Who has access" Anyone. Deploy and authorize.
      5. Copy the Web app URL and paste it above, click Save, then Test connection — it will now tell you plainly if the key doesn't match instead of just saying "Connected" when it isn't really working.
      6. The first synced result auto-creates a Results tab and a Dashboard tab. The script also rejects the wrong key, out-of-range scores, and duplicate submissions.
      7. If you already deployed a copy of this script and changed SYNC_KEY away from cEo0616, that's almost certainly why syncing looks "Connected" but nothing reaches your sheet: open Extensions → Apps Script on that sheet, change the SYNC_KEY line back to exactly cEo0616, save, then Deploy → Manage deployments → (pencil icon) → Version: New version → Deploy. This updates the same link already in the Reviewer, so you won't need to send students a new one.
      8. If you already had a Results sheet from before LRN was added, this new script upgrades it automatically the next time it runs — your existing rows are untouched, and an LRN column plus a new "current attempt only" Dashboard section are added going forward.
      9. A new "📊 Visual Summary" section (scorecards + two charts) is added near the bottom of the Dashboard tab automatically — scroll down to find it after your first sync with this version.

var SYNC_KEY = "cEo0616";
var MAX_FIELD_LENGTH = 300;
var EXPECTED_COLUMNS = 12;
var RESULTS_HEADERS = ["Timestamp","Student","Class","Lesson","Learning Competency","Score","Max","Percentage","Weakest Tag","Tag Breakdown","Recommendation","LRN"];
var ALLOWED_RECOMMENDATIONS = ["On Track","Needs Intervention","Ready for Advancement"];

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000); // wait up to 10s if another submission is being written
  } catch (lockErr) {
    Logger.log("LOCK TIMEOUT — could not acquire script lock");
    return jsonOut({ status: "error", message: "Server busy, please retry" });
  }
  try {
    var body = JSON.parse(e.postData.contents);
    Logger.log("Received row: " + JSON.stringify(body ? body.row : null) + " | id=" + (body ? body.id : null));
    if (!body || body.key !== SYNC_KEY) {
      Logger.log("REJECTED: key mismatch. Received key='" + (body ? body.key : null) + "', expected='" + SYNC_KEY + "'");
      return jsonOut({ status: "error", message: "Invalid sync key" });
    }
    if (!isValidRow(body.row)) {
      Logger.log("REJECTED: row failed isValidRow(). Row length=" + (body.row ? body.row.length : "n/a") + ", expected=" + EXPECTED_COLUMNS);
      return jsonOut({ status: "error", message: "Invalid data" });
    }
    if (isDuplicate(body.id)) {
      Logger.log("SKIPPED: id " + body.id + " already marked as synced (duplicate) — not appending again");
      return jsonOut({ status: "ok" });
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    Logger.log("Active spreadsheet: " + ss.getName() + " | " + ss.getUrl());
    var sheet = ss.getSheetByName("Results");
    if (!sheet) {
      Logger.log("No Results sheet found — creating it now");
      sheet = ss.insertSheet("Results");
      sheet.appendRow(RESULTS_HEADERS);
      sheet.getRange("M1").setValue("Is Latest");
      setupDashboard(ss);
    }
    ensureLrnColumn(sheet); // self-heals a Sheet created before LRN was added
    ensureLatestDashboardSection(ss); // self-heals a Dashboard created before LRN was added
    ensureAllTimeRecommendationFormula(ss); // self-heals the older pivot formula on existing sheets
    ensureVisualDashboard(ss); // self-heals a Dashboard created before charts/scorecards existed
    var cleanRow = sanitizeRow(body.row);
    sheet.appendRow(cleanRow);
    var newRowNumber = sheet.getLastRow();
    Logger.log("SUCCESS: row appended to sheet '" + sheet.getName() + "' at row " + newRowNumber);
    updateLatestFlags(sheet, cleanRow[11], cleanRow[3], newRowNumber);
    logSyncedId(body.id);
    return jsonOut({ status: "ok" });
  } catch (err) {
    Logger.log("EXCEPTION: " + err.message + " | stack: " + err.stack);
    return jsonOut({ status: "error", message: "Server error" });
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var resultsSheet = ss.getSheetByName("Results");
  var providedKey = e && e.parameter ? e.parameter.key : null;
  return jsonOut({
    status: "ok",
    message: "Reviewer results endpoint is live",
    keyValid: providedKey == null ? null : (providedKey === SYNC_KEY),
    spreadsheetName: ss.getName(),
    spreadsheetUrl: ss.getUrl(),
    resultsSheetExists: !!resultsSheet,
    resultsRowCount: resultsSheet ? resultsSheet.getLastRow() : 0
  });
}

function isValidRow(row) {
  if (!Array.isArray(row) || row.length !== EXPECTED_COLUMNS) return false;
  var score = Number(row[5]);
  var max = Number(row[6]);
  var pct = Number(row[7]);
  if (isNaN(score) || isNaN(max) || isNaN(pct)) return false;
  if (score < 0 || max <= 0 || score > max) return false;
  if (pct < 0 || pct > 100) return false;
  if (ALLOWED_RECOMMENDATIONS.indexOf(row[10]) === -1) return false;
  return true;
}

function sanitizeRow(row) {
  return row.map(function (v) {
    var s = (v === null || v === undefined) ? "" : String(v);
    s = s.trim();
    if (s.length > MAX_FIELD_LENGTH) s = s.substring(0, MAX_FIELD_LENGTH);
    return s;
  });
}

function isDuplicate(id) {
  if (!id) return false;
  var ids = JSON.parse(PropertiesService.getScriptProperties().getProperty("SYNCED_IDS") || "[]");
  return ids.indexOf(id) !== -1;
}

function logSyncedId(id) {
  if (!id) return;
  var props = PropertiesService.getScriptProperties();
  var ids = JSON.parse(props.getProperty("SYNCED_IDS") || "[]");
  ids.push(id);
  if (ids.length > 300) ids = ids.slice(ids.length - 300);
  props.setProperty("SYNCED_IDS", JSON.stringify(ids));
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

// Upgrades a Results sheet created before the LRN column existed —
// adds the LRN header and the "Is Latest" helper column without touching any existing rows.
function ensureLrnColumn(sheet) {
  var headerRange = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 1));
  var headers = headerRange.getValues()[0];
  if (headers.indexOf("LRN") === -1) {
    sheet.getRange(1, 12).setValue("LRN");        // column L
    sheet.getRange(1, 13).setValue("Is Latest");  // column M
  }
}

// Marks exactly one row as "Latest" for a given LRN+Lesson pair — clears any
// previous "Latest" flag for that same pair, then flags the newly appended row.
// Only ever touches the actual range of existing data rows, never open-ended,
// so it can never trick Sheets into treating unused rows as "used."
function updateLatestFlags(sheet, lrn, lessonTitle, newRowNumber) {
  var priorRowCount = newRowNumber - 2; // number of existing rows before this new one (row 2 is the first data row)
  if (priorRowCount > 0 && lrn) {
    var lrnValues = sheet.getRange(2, 12, priorRowCount, 1).getValues();
    var lessonValues = sheet.getRange(2, 4, priorRowCount, 1).getValues();
    for (var i = 0; i < lrnValues.length; i++) {
      if (lrnValues[i][0] === lrn && lessonValues[i][0] === lessonTitle) {
        sheet.getRange(i + 2, 13).setValue("");
      }
    }
  }
  sheet.getRange(newRowNumber, 13).setValue("Latest");
}

function setupDashboard(ss) {
  var dash = ss.getSheetByName("Dashboard");
  if (dash) return;
  dash = ss.insertSheet("Dashboard");
  dash.getRange("A1").setValue("Reviewer Dashboard").setFontWeight("bold").setFontSize(14);

  dash.getRange("A3").setValue("Average % by Lesson (all attempts, full history)");
  dash.getRange("A4").setFormula('=IFERROR(QUERY(Results!D2:H,"select D, avg(H) where D is not null group by D label avg(H) \'Avg %\'",0),"No data yet")');

  dash.getRange("D3").setValue("Recommendation counts by Lesson (all attempts, full history)");
  dash.getRange("D4").setFormula('=IFERROR(QUERY(Results!D2:K,"select D, count(E) where D is not null group by D pivot K",0),"No data yet")');

  dash.getRange("A14").setValue("Most common weak area by Lesson (top = most frequent)");
  dash.getRange("A15").setFormula('=IFERROR(QUERY(Results!D2:I,"select D, I, count(I) where D is not null and I is not null group by D, I order by count(I) desc",0),"No data yet")');

  dash.getRange("D14").setValue("Students: Needs Intervention (all attempts, full history)");
  dash.getRange("D15").setFormula('=IFERROR(QUERY(Results!B2:K,"select B, C, D, H where K = \'Needs Intervention\' order by D, H asc",0),"No data yet")');

  dash.getRange("H14").setValue("Students: Ready for Advancement (all attempts, full history)");
  dash.getRange("H15").setFormula('=IFERROR(QUERY(Results!B2:K,"select B, C, D, H where K = \'Ready for Advancement\' order by D, H desc",0),"No data yet")');

  writeLatestAttemptSection(dash);
  writeScorecards(dash);
  writeCharts(dash);
}

// Adds the "current status" section (based only on each student's MOST RECENT
// attempt per lesson, grouped by LRN) — self-heals a Dashboard created before this existed.
// Formulas are safe to rewrite every time (idempotent — no duplication risk like
// charts have), so this always refreshes rather than checking a marker. That means
// any future formula fix here applies automatically to every existing sheet on its
// next sync, instead of being silently skipped because "it was already written once."
function ensureLatestDashboardSection(ss) {
  var dash = ss.getSheetByName("Dashboard");
  if (!dash) return;
  writeLatestAttemptSection(dash);
}

function ensureAllTimeRecommendationFormula(ss) {
  var dash = ss.getSheetByName("Dashboard");
  if (!dash) return;
  dash.getRange("D4").setFormula('=IFERROR(QUERY(Results!D2:K,"select D, count(E) where D is not null group by D pivot K",0),"No data yet")');
}

function writeLatestAttemptSection(dash) {
  dash.getRange("A24").setValue("Average % by Lesson (current attempt only)");
  dash.getRange("A25").setFormula('=IFERROR(QUERY(Results!D2:M,"select D, avg(H) where D is not null and M = \'Latest\' group by D label avg(H) \'Avg %\'",0),"No data yet")');

  dash.getRange("D24").setValue("Current status by Lesson (current attempt only)");
  dash.getRange("D25").setFormula('=IFERROR(QUERY(Results!D2:M,"select D, count(E) where D is not null and M = \'Latest\' group by D pivot K",0),"No data yet")');

  dash.getRange("A35").setValue("Students currently needing intervention (based on their latest attempt)");
  dash.getRange("A36").setFormula('=IFERROR(QUERY(Results!B2:M,"select B, L, C, D, H where K = \'Needs Intervention\' and M = \'Latest\' order by D, H asc",0),"No data yet")');

  dash.getRange("F35").setValue("Students currently ready for advancement (based on their latest attempt)");
  dash.getRange("F36").setFormula('=IFERROR(QUERY(Results!B2:M,"select B, L, C, D, H where K = \'Ready for Advancement\' and M = \'Latest\' order by D, H desc",0),"No data yet")');
}

// Adds scorecards + charts in a fresh area (row 40+) so it never collides with
// anything already placed on an existing Dashboard — self-heals like the section above.
function ensureVisualDashboard(ss) {
  var dash = ss.getSheetByName("Dashboard");
  if (!dash) return;
  var marker = dash.getRange("A40").getValue();
  if (marker !== "📊 Visual Summary") {
    writeScorecards(dash);
  }
  writeCharts(dash); // always rebuilt fresh, so a future range/code fix applies automatically
}

function writeScorecards(dash) {
  dash.getRange("A40").setValue("📊 Visual Summary").setFontWeight("bold").setFontSize(13);

  dash.getRange("A42").setValue("Total Attempts").setFontWeight("bold").setFontSize(10);
  dash.getRange("A43").setFormula('=IFERROR(COUNTA(Results!D2:D),0)').setFontSize(24).setFontWeight("bold");

  dash.getRange("C42").setValue("Unique Students Reached").setFontWeight("bold").setFontSize(10);
  dash.getRange("C43").setFormula('=IFERROR(COUNTA(UNIQUE(FILTER(Results!L2:L,Results!L2:L<>""))),0)').setFontSize(24).setFontWeight("bold");

  dash.getRange("E42").setValue("Overall Avg % (current attempts)").setFontWeight("bold").setFontSize(10);
  dash.getRange("E43").setFormula('=IFERROR(ROUND(AVERAGE(FILTER(Results!H2:H,Results!M2:M="Latest")),1),"N/A")').setFontSize(24).setFontWeight("bold");

  dash.getRange("G42").setValue("Needing Intervention Now").setFontWeight("bold").setFontSize(10);
  dash.getRange("G43").setFormula('=IFERROR(COUNTIFS(Results!K2:K,"Needs Intervention",Results!M2:M,"Latest"),0)')
    .setFontSize(24).setFontWeight("bold").setFontColor("#B3564B");
}

// Charts are always removed and rebuilt from scratch on every call — this makes
// them self-correcting: if the ranges below are ever fixed again in the future,
// every already-existing sheet picks up that fix automatically on its next sync.
function writeCharts(dash) {
  var existingCharts = dash.getCharts();
  for (var i = 0; i < existingCharts.length; i++) {
    dash.removeChart(existingCharts[i]);
  }

  // Tightly bounded to rows 25-34 ONLY — row 24 is the section header (not data),
  // and row 35 is where the next, unrelated table begins. Never widen this past
  // row 34 without also checking what else occupies these columns below it.
  var avgChart = dash.newChart()
    .asColumnChart()
    .addRange(dash.getRange("A25:B34"))
    .setPosition(46, 1, 0, 0)
    .setOption("title", "Average % by Lesson (Current Attempt)")
    .setOption("legend", { position: "none" })
    .setOption("vAxis", { title: "Average %", minValue: 0, maxValue: 100 })
    .setOption("colors", ["#5B7A8C"])
    .setOption("width", 480)
    .setOption("height", 300)
    .build();
  dash.insertChart(avgChart);

  var statusChart = dash.newChart()
    .asColumnChart()
    .addRange(dash.getRange("D25:G34"))
    .setPosition(46, 6, 0, 0)
    .setOption("title", "Current Status by Lesson")
    .setOption("isStacked", true)
    .setOption("colors", ["#5BBE94", "#F08B72", "#FFB74D"])
    .setOption("width", 480)
    .setOption("height", 300)
    .build();
  dash.insertChart(statusChart);
}
